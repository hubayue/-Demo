const ContentCatalog = preload("res://src/content/content_catalog.gd")

    var file := FileAccess.open("res://tests/fixtures/v7.19.2-week-2948-keypoints.json", FileAccess.READ)

                return
    print("Godot v7.19.2 weekly levels: PASS")
    quit(0)

func _expect(condition: bool, message: String) -> bool:
    if condition:
        return true
    push_error(message)
    quit(1)
